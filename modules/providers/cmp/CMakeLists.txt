cmake_minimum_required(VERSION 3.16)
project(KotonohaProviderCMP LANGUAGES C CXX)
if(TARGET Kotonoha::ProviderCMP)
  return()
endif()
if(NOT TARGET Kotonoha::Engine)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../lib"
      "${CMAKE_CURRENT_BINARY_DIR}/kotonoha")
endif()
add_library(kotonoha_provider_cmp STATIC src/Pack.cpp src/SummerAssets.cpp)
add_library(Kotonoha::ProviderCMP ALIAS kotonoha_provider_cmp)
target_include_directories(kotonoha_provider_cmp PUBLIC include)
target_link_libraries(kotonoha_provider_cmp PUBLIC Kotonoha::Engine)
target_compile_features(kotonoha_provider_cmp PUBLIC cxx_std_17)
set_target_properties(kotonoha_provider_cmp PROPERTIES CXX_EXTENSIONS OFF)
