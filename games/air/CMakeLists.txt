cmake_minimum_required(VERSION 3.16)
project(Air VERSION 0.1.0 LANGUAGES C CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
endif()
add_library(AirFormats STATIC src/Images.cpp src/Voice.cpp src/Script.cpp src/Session.cpp src/System.cpp
  vendor/libreallive/archive.cc vendor/libreallive/bytecode.cc
  vendor/libreallive/compression.cc vendor/libreallive/expression.cc
  vendor/libreallive/filemap.cc vendor/libreallive/scenario.cc)
target_include_directories(AirFormats PUBLIC include vendor)
target_compile_features(AirFormats PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(AirFormats PUBLIC /J)
else()
  target_compile_options(AirFormats PUBLIC -funsigned-char)
endif()
if(NOT ANDROID)
  add_executable(AirInspect tools/AirInspect.cpp)
  target_link_libraries(AirInspect PRIVATE AirFormats)
endif()
if(NOT TARGET Kotonoha::Engine)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib" "${CMAKE_CURRENT_BINARY_DIR}/kotonoha")
endif()
add_library(air_port STATIC src/Config.cpp src/AirAssets.cpp src/AirRuntime.cpp)
target_include_directories(air_port PUBLIC src)
find_package(Iconv REQUIRED)
target_link_libraries(air_port PUBLIC AirFormats Kotonoha::Engine Iconv::Iconv)
if(BUILD_TESTING)
  foreach(format Images Voice Script Session System)
    add_executable(Air${format}Tests tests/${format}Test.cpp)
    target_link_libraries(Air${format}Tests PRIVATE AirFormats)
    if(format STREQUAL "Session" AND AIR_TEST_DATA)
      add_test(NAME Air${format} COMMAND Air${format}Tests "${AIR_TEST_DATA}/Seen.txt")
    else()
      add_test(NAME Air${format} COMMAND Air${format}Tests)
    endif()
  endforeach()
endif()

kotonoha_add_app(Air src/main.cpp)
target_link_libraries(Air PRIVATE air_port)
kotonoha_configure_app(Air BUNDLE_ID me.hirameki.kotonoha.air GAME air)
if(BUILD_TESTING)
  add_executable(AirEngineTests tests/EngineTest.cpp)
  target_link_libraries(AirEngineTests PRIVATE air_port)
  add_executable(AirConfigTests tests/ConfigTest.cpp)
  target_link_libraries(AirConfigTests PRIVATE air_port)
  add_test(NAME AirConfig COMMAND AirConfigTests)
  add_executable(AirTitleInputTests tests/TitleInputTest.cpp)
  target_link_libraries(AirTitleInputTests PRIVATE air_port)
  add_test(NAME AirTitleInput COMMAND AirTitleInputTests)
endif()

set(AIR_TEST_DATA "" CACHE PATH "AIR Standard Edition original data for integration tests")
if(BUILD_TESTING)
  set_tests_properties(AirSession PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME AirEngine COMMAND AirEngineTests "${AIR_TEST_DATA}")
  set_tests_properties(AirEngine PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" SKIP_RETURN_CODE 77 TIMEOUT 90 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
  if(AIR_TEST_DATA)
    add_test(NAME AirRetailImages COMMAND AirImagesTests "${AIR_TEST_DATA}/g00" "${AIR_TEST_DATA}/pdt")
    add_test(NAME AirRetailVoice COMMAND AirVoiceTests "${AIR_TEST_DATA}")
    add_test(NAME AirRetailScripts COMMAND AirScriptTests "${AIR_TEST_DATA}/Seen.txt")
    add_test(NAME AirRetailSession COMMAND AirSessionTests "${AIR_TEST_DATA}/Seen.txt")
  endif()
endif()
