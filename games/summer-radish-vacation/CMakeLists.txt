cmake_minimum_required(VERSION 3.16)
project(SummerRadishVacation VERSION 0.1.0 LANGUAGES C CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
add_library(SummerRadishFormats STATIC src/Pack.cpp src/Script.cpp src/Map.cpp src/Transition.cpp src/Native.cpp)
target_include_directories(SummerRadishFormats PUBLIC include)
target_compile_features(SummerRadishFormats PUBLIC cxx_std_17)
if(NOT TARGET Kotonoha::Engine)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib" "${CMAKE_CURRENT_BINARY_DIR}/kotonoha")
endif()
add_library(summer_radish_vacation STATIC src/SummerAssets.cpp src/SummerRuntime.cpp)
target_include_directories(summer_radish_vacation PUBLIC src)
target_link_libraries(summer_radish_vacation PUBLIC Kotonoha::Engine SummerRadishFormats)
kotonoha_add_app(SummerRadish src/main.cpp)
target_link_libraries(SummerRadish PRIVATE summer_radish_vacation)
kotonoha_configure_app(SummerRadish BUNDLE_ID me.hirameki.kotonoha.summerradishvacation GAME summer-radish-vacation)
if(NOT ANDROID)
    add_executable(SummerInspect tools/SummerInspect.cpp)
    target_link_libraries(SummerInspect PRIVATE SummerRadishFormats)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
